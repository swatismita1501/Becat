proc chkfile(var proc_err String, cf String) Logical
	Var
		fs fileSystem
	endVar

	; 1. Check for required files
	message(sname, "Checking for " + cf)
	; Search for cf
	if not fs.findFirst(cf) then
		proc_err = proc_err + "\n" + sname + cf + " not found."
		message(sname, " File " + cf + " not found!")
		return FALSE
	else
		return TRUE
	endif

endProc

; Process uses Table delete for .DB files so that all associated
; files will be deleted. 
proc FileDel(fileaccess String)
  var
	  fs FileSystem
	  srctb1 Table
	  pname String
	endVar
	pname = "FileDel:"
	if not isignoreCaseInStringCompares() then
		ignoreCaseInstringCompares(Yes)
	endIf
	if fs.findFirst(fileaccess) then
		message(selected, sname, pname, " Deleting ", fileaccess)
		if advMatch(fileaccess, ".db$") then
			srctb1.attach(fileaccess)
			srctb1.delete()		
		else
			fs.delete(fileaccess)
		endIf
	endIf
	ignoreCaseInstringCompares(No)
endproc

; Process to clean up a directory and then enter it into
; DELDIRS.DB for deletion during a new session of Paradox.
; Bug in Paradox 7.0 prevents deletion of a directory accessed
; by Paradox during that session. 
proc DirectoryDel(dirtodel String)
	var
		fileName			String
		fs					fileSystem
		pname 			String
		deldirs			String
		tcdirs			TCursor
	endVar
	pname = "DirectoryDel:"
	deldirs = workingdir() + "\\DELDIRS.DB"
	errorTrapOnWarnings(NO)
	; Delete files
	if fs.findFirst(dirtodel + "\\*.*") then
		fileName = fs.name()
		message(selected, sname, pname, " Deleting ", fileName)
		if fileName <> "." and fileName <> ".." then
			fs.delete(dirtodel + "\\" + fileName)
		endIf				
		while fs.findNext()  
			fileName = fs.name()
			message(selected, sname, pname, " Deleting ", fileName) 
			if fileName <> "." and fileName <> ".." then
				fs.delete(dirtodel + "\\" + fileName)
			endIf				
		endWhile
	endIf
	; Enter directory to be deleted if actually a directory.
	if fs.findFirst(dirtodel + "\\*.*") then
		message(selected, sname, pname, " Saving ", dirtodel, " for future deletion.")
		tcdirs.open(deldirs)
		tcdirs.end()
		tcdirs.edit()
		tcdirs.insertAfterRecord()
		tcdirs.setFieldValue("Name", dirtodel)
		tcdirs.setFieldValue("Date", bldDate)
		tcdirs.endEdit()
		tcdirs.close()
	endIf
	errorTrapOnWarnings(YES)
endProc

; Process uses Table copy for .DB files so that all associated
; files will be copied. 
proc copyfile(expect String, filen String, fromdir String, todir String)
	var
		fs fileSystem
		srctb1 Table
	endvar
	if not isignoreCaseInStringCompares() then
		ignoreCaseInstringCompares(Yes)
	endIf
	expect = expect.upper()
	if expect = "Y" then
		if fs.findFirst(fromDir + filen) then
			message(selected, sname, "Copying ", filen, " to " + todir + ".")
			if advMatch(filen, ".db$") then
				srctb1.attach(fromDir + filen)
				if not srctb1.copy(todir + filen) then
					msgInfo("", "Unable to copy " + filen)
				endIf			
			else
				if not fs.copy(fromDir + filen, toDir + filen) then
					msgInfo("", "Unable to copy " + filen)
				endIf
			endIf
		else
			msgInfo("", filen + " not found to copy.")
		endIf
	endIf
	ignoreCaseInstringCompares(No)
endProc

{
 Process to create and call a batch file.
 todo : CE to create and execute; C to create; A to append; AE to append and execute.
 batcall : command line to be executed.
 05/09/97 Modified and made a standard process in order to handle ampersands in the
			 command line.
 }
proc callbat(todo string, batcall string)
	var
		batfile	String
		tsbat TextStream
		n, n1 longInt
		ar Array[] String
	endVar
	batcall.breakApart(ar, "&")
	n = ar.size()
	batcall = ar[1]
	for n1 from 2 to n
		batcall = batcall + "^&" + ar[n1] 
	endFor
	batfile = workingDir() +  "\\TEMP.BAT"
	switch
		case todo.match("C..") = TRUE : 
			tsbat.create(batfile)
			tsbat.WriteLine(batcall)
			tsbat.close() 
		case todo.match("A..") = TRUE : 
			tsbat.open(batfile, "a")
			tsbat.WriteLine(batcall)
			tsbat.close() 
	endSwitch
	if todo.match("..E") = TRUE then
		execute(batfile, TRUE)
		sleep(5000)
		bringToTop()
		show()
	endIf
endProc

proc BldHstReport(procdesc String, inf String, outf String)

	var
	  time_in_sec_complete, time_in_sec_start LongInt
	  time_elapsed, rec_per_min Number
	  hr_start, hr_complete, min_start, min_complete, sec_start, sec_complete LongInt

	  tc1pr Tcursor
	  srcTb1 Table

	  pname, nrec String
	endVar
	pname = "BldHstReport:"

   ; 1. Insert record for current build into BLD_HST
	message(selected, sname, pname, " Inserting build data into ", histf, ".")

	tc1pr.open(histf)
   lock(tc1pr, "Exclusive")
	tc1pr.edit()
   tc1pr.end()
	tc1pr.insertAfterRecord()
   procdesc = procdesc.subStr(1,45)
	tc1pr.setFieldValue("Process", procdesc)
   tc1pr.setFieldValue("Date Processed", blddate)
   tc1pr.setFieldValue("Cat_Month", cat_month)
	tc1pr.setFieldValue("Time_Started", time_start)
   tc1pr.setFieldValue("Time_complete", time_complete)
   hr_start = time_start.hour()
	hr_complete = time_complete.hour()
   min_start = time_start.minute()
   min_complete = time_complete.minute()
	sec_start = time_start.second()
   sec_complete = time_complete.second()
   time_in_sec_complete = (hr_complete * 3600) + (min_complete * 60) + sec_complete
	time_in_sec_start = (hr_start * 3600) + (min_start * 60) + sec_start
   if time_in_sec_complete < time_in_sec_start then
      time_elapsed = (time_in_sec_complete + 86400 - time_in_sec_start)/60
	else
      time_elapsed = (time_in_sec_complete - time_in_sec_start)/60
   endIf
	nrec = strVal(time_elapsed)
   tc1pr.setFieldValue("Time_Elapsed", time_elapsed)
   if time_elapsed = 0 then
	  time_elapsed = .01
   endIf
   rec_per_min = cat_rec_proc/time_elapsed
	nrec = strVal(rec_per_min)
   tc1pr.setFieldValue("rec_per_min", nrec)
   srctb1.attach(outf)
	nrec = strVal(srctb1.nRecords())
   tc1pr.setFieldValue("Cat_Recs", nrec)
   srctb1.attach(inf)
	nrec = strVal(srctb1.nRecords())
   tc1pr.setFieldValue("Inpt_Recs", nrec)
	nrec = strVal(cat_rec_proc)
   tc1pr.setFieldValue("Cat_Rec_Proc", nrec)
	nrec = strVal(inpt_rec_proc)
   tc1pr.setFieldValue("Imp_Rec_Proc", nrec)
	nrec = strVal(cat_rec_changed) 
   tc1pr.setFieldValue("Cat_Rec_Changed", nrec)
	tc1pr.endedit()
   unlock(tc1pr, "Exclusive")
	tc1pr.close()

endProc

proc remLzero(strValue String) String

var
 nstrValue String
endvar

  while strValue.match("0..", nstrValue)
	 strValue = nstrValue
  endWhile
  return strValue

endproc

proc deDash(strValue String) String

var
 nstrValue, nstr1Value, nstr2Value String
endvar

  while strValue.match("-..", nstrValue)
    strValue = nstrValue
  endWhile
  while strValue.match("..-", nstrValue)
    strValue = nstrValue
  endWhile
  while strValue.match("..-..", nstr1Value, nstr2Value)
    strValue = nstr1Value + nstr2Value
  endWhile
  return strValue

endproc

;--------------------------------------------------------

proc deSpace(strValue String) String

var
 nstrValue, nstr1Value, nstr2Value String
endvar

  while strValue.match(" ..", nstrValue)
    strValue = nstrValue
  endWhile
  while strValue.match(".. ", nstrValue)
    strValue = nstrValue
  endWhile
  while strValue.match(".. ..", nstr1Value, nstr2Value)
    strValue = nstr1Value + nstr2Value
  endWhile
  return strValue

endproc

;
; Procedure to pad string variable space
;
proc spacepadStr(MaxS smallInt, FieldN String, typePad String) String
	 var
       padSize smallInt
		 bSpace, padSpace String
	 endVar

	 bSpace = "                                                                           "
	 padSize = MaxS - FieldN.size() 
	 padSpace = bspace.substr(1,padSize)        
	 if typePad = "L" then
		return  (padSpace + FieldN )
	 else
		return  (FieldN + padSpace)
	 endif
endproc

;--------------------------------------------------------
;
; Procedure to pad string variable space
;
proc zeropadStr(MaxS smallInt, FieldN String, typePad String) String
    var
       padSize smallInt
       bSpace, padSpace String
    endVar

    bSpace = "0000000000000000"
    padSize = MaxS - FieldN.size() 
    padSpace = bspace.substr(1,padSize)        
    if typePad = "L" then
      return  (padSpace + FieldN )
    else
      return  (FieldN + padSpace)
    endif
endproc

;
; Procedure to derive manufacturer part number from description
;
proc partFDesc(desc String) String
    var
       locBlank, mpartLen, locTotal SmallInt
       mpart String
    endVar
    if desc <> "" then
       locBlank = desc.search(" ")
       if locBlank = 0 then
          locBlank = 15
       endif
       mpartLen = min(locBlank, 15) - 1
       mpart = desc.substr(1, mpartLen)
       ;Check for lack of numbers
       locTotal = mpart.search("0") + mpart.search("1") + mpart.search("2") + mpart.search("3") +
                  mpart.search("4") + mpart.search("5") + mpart.search("6") + mpart.search("7") +
                  mpart.search("8") + mpart.search("9")
       if locTotal = 0 then
          mpart = ""
       endif
       return mpart
     else
       return ""
     endIf
endproc

;
; Procedure: If positive numeric right justify and zero fill;
; else put in default. No sign,
;
proc intfill(MaxS smallInt, FieldN String, defnum String) String
    var
		 inttest, maxI longInt
	 endVar
	 switch
		 case maxS = 1 : maxI = 9
		 case maxS = 2 : maxI = 99
		 case maxS = 3 : maxI = 999
		 case maxS = 4 : maxI = 9999
		 case maxS = 5 : maxI = 99999
		 case maxS = 6 : maxI = 999999
		 case maxS = 7 : maxI = 9999999
		 case maxS = 8 : maxI = 99999999
		 otherwise : msgInfo("STOP","Unsupported option infill - maxS.")
                   return ""
	 endSwitch
    if (FieldN <> "") then
       FieldN = FieldN.ltrim()
       FieldN = remLzero(FieldN)
		 try inttest = longInt(FieldN)
			switch
				case inttest >= 1 and inttest < maxI :
					FieldN = zeropadStr(MaxS, FieldN, "L")
				case inttest >= 1 and inttest >= maxI :
					FieldN = strVal(maxI)
				otherwise :
					FieldN = defnum
			endSwitch
       onFail
          FieldN = defnum
       endTry
    else
       FieldN = defnum
    endIf
    return FieldN
endproc
;--------------------------------------------------------
;
; Procedure: 
;   If numeric according to UTNMPX, left justify; ***Catalog special processing***
;                                   left pad zero if logical zeropadnumerics = Y
;                                   keep or remove left zeros on logical keepzeros
;   If ascii, leftjustify
proc catnumascii(MaxS smallInt, FieldN String, keepzeros Logical, zeropadnumerics Logical) String
	 var
		 tryField	String
		 InputDecimal	SmallInt
		 OutputDecimal	SmallInt
		 OutString		String
		 size smallInt
	 endVar
	 InputDecimal	= 0
	 OutputDecimal	= 0
	 OutString 		= "XXXXXXXXXXXXXX"   ; Must be 13 characters

	 if (FieldN <> "") then
		 FieldN = FieldN.ltrim()
		 size = fieldN.size()
		 if size > maxS then
			 FieldN = fieldN.subStr(1, maxS)
		 endIf
		 ; The test field given to UTNMPX must be completely filled since all Cobol 
		 ; fields are a fixed size.
		 tryField = tosize(MaxS, FieldN)
		 if UTNMPX(TryField, MaxS, InputDecimal, OutputDecimal, OutString) <> 0 Then
		 ;Field is numeric
			 if zeropadnumerics then
				 FieldN = zeropadStr(MaxS, FieldN, "L")
			 else
				 if not keepzeros then
					FieldN = FieldN.ltrim()
					FieldN = remLzero(FieldN)
				 endIf
;				 FieldN = spacepadStr(MaxS, FieldN, "L")
			 endIf
		 else
		 ;Field is ascii
			 FieldN = FieldN.ltrim()
		 endIf
	 endIf
	 return FieldN
endproc
;--------------------------------------------------------
; Procedure: Limit string to correct size. Pad with spaces.
proc tosize(maxS smallInt, fieldN String) String
	var
		size smallInt
	endVar
	size = fieldN.size()
	switch
		case size > maxS :
			 fieldN = fieldN.subStr(1, maxS)
		case size < maxS :
			 fieldN = spacepadStr(MaxS, FieldN, "R")
	endSwitch
	return fieldN 
endProc

endproc

;--------------------------------------------------------
; Procedure: tascii: Fetch ascii field; If filled, left justify; if blank, fill
;            with default; Write and return corrected field.
proc tascii(Var tcin TCursor, Var tcout TCursor, MaxS smallInt, FieldName String, default String) String
    var
       i smallInt
       FieldN String
    endVar
    tcin.fieldValue(FieldName, FieldN)
    if (FieldN <> "") then
       FieldN = FieldN.ltrim()
       i = FieldN.size()
       if i > MaxS then
          FieldN = FieldN.subStr(1, MaxS)
       endIf
    else
       FieldN = default
    endIf
    tcout.setFieldValue(FieldName, FieldN)
    return FieldN
endproc

; Procedure: castNumCobol: Round, format and prepare a numeric value for
;                          storage in a Cobol numeric format.
proc castNumCobol(MaxS smallInt, DecimalPlace smallInt, n Number) String
    var
       FieldN, fieldSign String
       ntmp Number
       maxPlace smallInt
       maxN Number
    endVar
    if DecimalPlace > MaxS then
       msgInfo("STOP","castNumCobol - call incorrect.") 
    endIf
    maxPlace = (MaxS - DecimalPlace)
    fieldSign = "+"
    switch
       case maxPlace = 0 : maxN = .9999
       case maxPlace = 1 : maxN = 9.9999
       case maxPlace = 2 : maxN = 99.9999
       case maxPlace = 3 : maxN = 999.9999
       case maxPlace = 4 : maxN = 9999.9999
       case maxPlace = 5 : maxN = 99999.9999
       otherwise : msgInfo("STOP","Unsupported option castNumCobol - maxPlace.")
                   return ""
    endSwitch
    if n < 0 then
       n = n.abs()
       fieldSign = "-"
    endIf
    if n > maxN then
       n = maxN
       switch
          case DecimalPlace = 1 : nTmp = n.truncate(1)*10
          case DecimalPlace = 2 : nTmp = n.truncate(2)*100
          case DecimalPlace = 3 : nTmp = n.truncate(3)*1000
          case DecimalPlace = 4 : nTmp = n.truncate(4)*10000
          otherwise : msgInfo("STOP","Unsupported option castNumCobol - DecimalPlace.")
                      return ""
       endSwitch
    else     
       switch
          case DecimalPlace = 1 : nTmp = n.round(1)*10
          case DecimalPlace = 2 : nTmp = n.round(2)*100
          case DecimalPlace = 3 : nTmp = n.round(3)*1000
          case DecimalPlace = 4 : nTmp = n.round(4)*10000
          otherwise : msgInfo("STOP","Unsupported option castNumCobol - DecimalPlace.")
                      return ""
       endSwitch
    endIf
    FieldN = format("w" + StrVal(MaxS) + ".0", nTmp)
    FieldN = fieldN.lTrim()
    FieldN = fieldSign + zeropadStr(MaxS, fieldN, "L")
    return FieldN
endproc

proc massageUpc(upc String) String
	var
		stringtest, zeros String
		numtest Number
		size, zerosize smallInt
	endvar
	zeros = "000000000000"
	upc = upc.lTrim()
	try numtest = numval(upc)
		stringtest = remLzero(upc)
		size = stringtest.size()
		switch
			case size = 0  : upc = ""
			case size > 6  : upc = spacepadStr(12, upc, "L")    
			otherwise      :
				zerosize = 6 - size
				upc = zeros.substr(1, zerosize) + stringtest 
				upc = spacepadStr(12, stringtest, "L")
		endSwitch
	onFail
		upc = ""
	endTry
	return upc        
endProc
;--------------------------------------------------------
; Procedure: addcentury: Covert two digit year to four digit
;            year
; 9/27/99 - Added code to only add a century if the year is
;           between 0 and 100. Will not add century if the
;           the year already includes century. MLH
proc addcentury(year smallInt) smallInt
	if year < 100 then
		if year < 80 then
			year = year + 2000
		else
			year = year + 1900
		endIf
   endIf
	return year
endProc

proc ReformatDate(tgtdate Date) String
; See addcentury for year format yy or yyyy code check.
	var
		year String
		datearray Array[] String
		yearInt smallInt
	endVar
	tgtdate.breakApart(datearray, "/")
	if datearray[1].size() = 1 then
		datearray[1] = "0" + datearray[1]
	endIf
	if datearray[2].size() = 1 then
		datearray[2] = "0" + datearray[2]
	endIf
	yearInt = smallInt(datearray[3])
	yearInt = addcentury(yearInt)
   ; YearInt here always includes century
   ; Therefore StrVal will have format yyyy
	year = strVal(yearInt)
	return year + datearray[1] + datearray[2]

endProc

; For Standard Leading and Trailing processing. Problem is that loadtape fineline 
; processing zero fills all fineline codes and there is no way to know how
; many zeros have been added. Thus this procedure will be wholesaler specific until
; the loadtape fineline processing is rewritten to output a special file for Eagle
; catalog builds. 7/27/98 MLH
; Problem for Orgil is really that in order to make the drill down work two leading
; zeros must be eliminated. Class code is 3 digits; fineline code is only 4 digits.
; Data is coming in with leading zeros. Loadtape would be adding them if they were
; not coming in that way. 7/27/98 MLH
; Orgil fineline codes: Last four digits will enable drill down. First two digits 
; usually "00" however can be "01" or "06"; not clear what first two digits stand
; for but will eliminate them anyway and clear with Product Development. 7/27/98 MLH
; 3/21/2003 Orgil now is sending in 5 digit fineline that drills down correctly. MLH
; 12/17/2009 See comment below. MLH
; 4/17/2012 Modified for JEN catalog. MLH
; 3/6/2014 Modified for UNI catalog. MLH
; INFO: this code corrects the item_raw fineline and the LTFI fineline so that matches can be found.
;
proc NormFinecode(loadcode string) string
	var
		i smallInt
		firstascii, newcode String
	endVar
		; In case fineline code happens to be blank.
		if loadcode = "" then
			return ""
		endIf
		switch
			case WhlCode = "ORGIL"	:
				i = loadcode.size()
				; Orgill's fineline is always 6 characters in ltfi as well as in item raw but
				; to make it match a class, the leading 0 needs to be removed.
				if i = 6 then
;               newcode = loadcode.substr(3, 4) Rewritten 3/21/2003
					newcode = loadcode.substr(2, 5)
				else
					newcode = ""
				endIf
			case WhlCode = "HWI"		:
				i = loadcode.size()
				firstascii = loadcode.subStr(1,1)
				if i <> 0 and firstascii = "0" then
					newcode = loadcode.substr(3, i-2)
				else
					newcode = loadcode
				endIf
			case WhlCode = "EMERY"	:
				; Emery's fineline code in item raw will be four characters long and match to a class.
				; Emery's fineline code in ltfi with be six characters long and match to a class only
				; if the leading two zeros are removed. There two leading zeros because their four character
				; fineline code has been zero filled left in a six character field for loading into
				; the Eagle. This may be a mistake in the Loadtape companion file software - I don't know.
				i = loadcode.size()
				switch
					case i = 6 :
						newcode = loadcode.substr(3, 4)
					case i = 4 :
						newcode = loadcode
				endSwitch
			case WhlCode = "JEN" :
				; Jensen's fineline code is four characters long.
				; LTFI fineline code has been zero filled left in the six character field. Therefore left most 00 needs to be
				; removed to get a match.
				i = loadcode.size()
				switch
					case i = 6 :
						newcode = loadcode.substr(3, 4)
					case i = 4 :
						newcode = loadcode
 				endSwitch
			case WhlCode = "UNI" :
				; United's fineline code is five characters long.
				; United's class code was created from the fineline code; as a result drill down from
            ; class to fineline should work.
				; LTFI fineline code has been zero filled left in the six character field. Therefore left most 0 needs to be
				; removed to get a match.
				i = loadcode.size()
				switch
					case i = 6 :
						newcode = loadcode.substr(2, 5)
					case i = 5 :
						newcode = loadcode
				endSwitch

			otherwise					:
				newcode = loadcode
		endSwitch
		return newcode
endProc

proc checkGTIN(GTIN String) Logical

	var
		Gtin1, Gtin2, Gtin3, Gtin4, Gtin5, Gtin6,
      Gtin7, Gtin8, Gtin9, Gtin10, Gtin11, Gtin12,
      Gtin13, Gtin14 Number
		GTIN_Odd_Total, GTIN_Even_Total, GTIN_Total, GTIN_Chk_Digit, numChk Number
		lenGTIN Smallint
	endvar

	; Check GTIN
	if GTIN = "00000000000000" or GTIN = "" then
		return False
	endIf
	lenGTIN = GTIN.size()
	if lenGTIN <> 14 then
		return False
	endIf
	try
		numChk = numVal(GTIN)
	onfail
		return False
	endtry

	Gtin1 = numVal(GTIN.subStr(1, 1))
	Gtin3 = numVal(GTIN.subStr(3, 1))
	Gtin5 = numVal(GTIN.subStr(5, 1))
	Gtin7 = numVal(GTIN.subStr(7, 1))
	Gtin9 = numVal(GTIN.subStr(9, 1))
	Gtin11 = numVal(GTIN.subStr(11, 1))
	Gtin13 = numVal(GTIN.subStr(13, 1))
	GTIN_Odd_Total = 3*(Gtin1+Gtin3+Gtin5+Gtin7+Gtin9+Gtin11+Gtin13)
	Gtin2 = numVal(GTIN.substr(2, 1))
	Gtin4 = numVal(GTIN.substr(4, 1))
	Gtin6 = numVal(GTIN.substr(6, 1))
	Gtin8 = numVal(GTIN.substr(8, 1))
	Gtin10 = numVal(GTIN.substr(10, 1))
	Gtin12 = numVal(GTIN.substr(12, 1))
	GTIN_Even_Total = Gtin2+Gtin4+Gtin6+Gtin8+Gtin10+Gtin12
	GTIN_Total = GTIN_Odd_Total+GTIN_Even_Total
	GTIN_Chk_Digit = Round((GTIN_Total+5), -1) - GTIN_Total
	If GTIN_Chk_Digit = 10 Then
		GTIN_Chk_Digit = 0
	EndIf
	if (GTIN_Chk_Digit <> Numval(GTIN.substr(14, 1))) then
		return False
	endif
	return True

endproc

      Gtin13, Gtin14 Number

	Gtin1 = numVal(GTIN.subStr(1, 1))
	Gtin3 = numVal(GTIN.subStr(3, 1))
	Gtin5 = numVal(GTIN.subStr(5, 1))
	Gtin7 = numVal(GTIN.subStr(7, 1))
	Gtin9 = numVal(GTIN.subStr(9, 1))
	Gtin11 = numVal(GTIN.subStr(11, 1))
	Gtin13 = numVal(GTIN.subStr(13, 1))
	GTIN_Odd_Total = 3*(Gtin1+Gtin3+Gtin5+Gtin7+Gtin9+Gtin11+Gtin13)
	Gtin2 = numVal(GTIN.substr(2, 1))
	Gtin4 = numVal(GTIN.substr(4, 1))

	Gtin8 = numVal(GTIN.substr(8, 1))

			mcost = intfill(8, mcost, "00000000")

; Add STDL/STDT UPC data to Eagle Item catalog file
proc stdUPCIntoOut(errf String, upcskuf String, kvf String, upcf String, ItemFinalDb String)

	var
	  sku, sku_cmp, newupc, newgtin String

	  tckv, tcerr, tccat, tcupc Tcursor
	  srcTb1 Table
	  nrec, pname String
	endVar
	pname = "stdUPCIntoOut:"


	; B. Sort sku file table by sku
	message(selected, sname, pname, "Sorting and indexing ", upcskuf, " by sku.")
	srctb1.attach(upcskuf)	
	sort srctb1
		on "Whl_SKU"
	endSort
	index srctb1
		PRIMARY
		on "Whl_SKU"
	endIndex

	srctb1.dropIndex()
	if isTable(":PRIV:KEYVIOL.DB") = TRUE then
		srctb1.attach(":PRIV:KEYVIOL.DB")
		srctb1.copy(kvf)
	endIf

	; C. Insert key violation data in error file
	if istable(kvf) then
		tckv.open(kvf)
      tcerr.open(errf)
		lock(tckv, "Exclusive", tcerr, "Exclusive")
		tcerr.edit()
		tcerr.end()
		tckv.home() 
		scan tckv:
         tckv.fieldValue("Whl_Sku", sku)
			tcerr.insertAfterRecord()
         tcerr.setFieldValue("TypeCode", "SKU")
			tcerr.setFieldValue("RefCode", sku.ltrim())
			tcerr.setFieldValue("Problem", "Duplicate SKU UPC file")
      endScan
		tcerr.endedit()
		unlock(tckv, "Exclusive", tcerr, "Exclusive")
      tckv.close()
		tcerr.close()
	endIf

	; B. Sort upc file table by sku
	message(selected, sname, pname, "Sorting and indexing ", upcf, " by sku.")
	srctb1.attach(upcf)	
	sort srctb1
		on "Whl_SKU"
	endSort

	; D. Insert upc data into item catalog file
   srctb1.attach(ItemFinalDb)
	nrec = strVal(srctb1.nRecords())
	srctb1.unAttach()

	tccat.open(ItemFinalDb)
	tcupc.open(upcf)
	tcerr.open(errf)
	lock(tccat, "Exclusive", tcupc, "Exclusive", tcerr, "Exclusive")
	tccat.edit()
	tcerr.edit()
	tccat.home()
	tcupc.home()
	tcerr.end()
	cat_rec_proc = 0
	inpt_rec_proc = 0
	cat_rec_changed = 0
	cat_rec_spec = 0
	scan tccat:
		tccat.fieldValue("Whl_Sku", sku)
		tcupc.fieldValue("Whl_Sku", sku_cmp)
      cat_rec_proc = cat_rec_proc + 1
		while (sku_cmp < sku and tcupc.eot() = False)
         tcupc.nextRecord()
			tcupc.fieldValue("Whl_Sku", sku_cmp)
			inpt_rec_proc = inpt_rec_proc + 1
		endWhile
		if sku = sku_cmp then
			message(selected, sname, pname," CatRec# ", tccat.recNo(), " of ", nrec, ". SKU ", sku, ".")

			; LR Whl_UPC X(13)
			newupc = tcupc."Whl_UPC"
			newupc = massageUpc(newupc)
			if newupc <> "" then

				; Whl_UPC X(13)
				; LT Whl_UPC X(13)
				tccat.setFieldValue("Whl_UPC", newupc)
				cat_rec_changed = cat_rec_changed + 1

			else

				; Report Not In File
				tcerr.insertAfterRecord()
				tcerr.setFieldValue("TypeCode", "SKU")
				tcerr.setfieldValue("RefCode", sku)
				tcerr.setfieldValue("Problem", "Bad UPC for SKU")

				;cat_rec_spec = cat_rec_spec + 1
			endIf
         ; GTIN X(14)
         newgtin = tcupc."GTIN"
         newgtin = massageGTIN(newgtin)
         tccat.setFieldValue("GTIN", newgtin)
         if newGTIN = "00000000000000" then
				; Report Not In File
				tcerr.insertAfterRecord()
				tcerr.setFieldValue("TypeCode", "SKU")
				tcerr.setfieldValue("RefCode", sku)
				tcerr.setfieldValue("Problem", "Bad GTIN for SKU")
            cat_rec_spec = cat_rec_spec + 1
			endIf
		else

			tcerr.insertAfterRecord()
			tcerr.setFieldValue("TypeCode", "SKU")
			tcerr.setfieldValue("RefCode", sku)
			tcerr.setfieldValue("Problem", "NIF UPC / GTIN for SKU")
			cat_rec_spec = cat_rec_spec + 1
		endIf
		sleep(0)
	endScan
	tccat.endedit()
	tcerr.endedit()
	unlock(tccat, "Exclusive", tcupc, "Exclusive", tcerr, "Exclusive")
	tccat.close()
	tcupc.close()
	tcerr.close()

endproc

	  srcTb1 Table

	tcscode.close()

	; B. Sort scode file table by sku
	message(selected, sname, pname, "Sorting and indexing ", scodef, " by sku.")
	srctb1.attach(scodef)	
	sort srctb1
		on "SKU"
	endSort

	; C. Insert scode data into item catalog file
	srctb1.attach(ItemFinalDb)
	nrec = strVal(srctb1.nRecords())
	srctb1.unAttach()

	tccat.open(ItemFinalDb)
	tcscode.open(scodef)
	lock(tccat, "Exclusive", tcscode, "Exclusive")
	tccat.edit()
	tccat.home()
	tcscode.home()
	cat_rec_proc = 0
	inpt_rec_proc = 0
	cat_rec_changed = 0
	cat_rec_spec = 0
	scan tccat:
		tccat.fieldValue("Whl_Sku", sku)
		tcscode.fieldValue("SKU", sku_cmp)
		cat_rec_proc = cat_rec_proc + 1
		while (sku_cmp < sku and tcscode.eot() = False)
			tcscode.nextRecord()
			tcscode.fieldValue("SKU", sku_cmp)
			inpt_rec_proc = inpt_rec_proc + 1
		endWhile
		if sku = sku_cmp then
			message(selected, sname, pname," CatRec# ", tccat.recNo(), " of ", nrec, ". SKU ", sku, ".")

			; Long_Description_A
			; Not supplied; Fill first 70 bytes with short description 
			;               Fill last 50 bytes with DA wholesaler's RPM data.
			sensitivity = tcscode."W_S_CODE"
			if sensitivity = "" then
				sensitivity = " "
			endIf
			category = tcscode."CATEGORY"
			category = spacepadStr(4, category, "R")
			bmret = tcscode."BM_RETAIL"
			switch
				case bmret = "00000000" :
					bmretfmt = "         "
				otherwise :
					bmretpredec = bmret.subStr(1,5)
					switch
						case bmretpredec = "00000" : bmretfmt = "     " + "." + bmret.subStr(6,3)
						otherwise :
							bmretpredecint = smallInt(bmretpredec)
							bmretprefmt = strVal(bmretpredecint)
							bmretprefmt = spacepadStr(5, bmretprefmt, "L")
							bmretfmt = bmretprefmt + "." + bmret.subStr(6,3)
					endSwitch
			endSwitch

			desc = tccat."Long_Description_A"
			desc = spacepadStr(70, desc, "R")
			scodedesc = "PS Code: " + sensitivity + " Assortment: " + category + " Benchmark: " + bmretfmt
			desc = desc + scodedesc
			tccat.setFieldValue("Long_Description_A", desc)

			cat_rec_changed = cat_rec_changed + 1

		endIf
		sleep(0)
	endScan
	tccat.endedit()
	unlock(tccat, "Exclusive", tcscode, "Exclusive")
	tccat.close()
	tcscode.close()

endproc

;|BeginMethod|#Library1|C4_classproc|
;  C4_classproc  Process LTCL data into ecat class file. Created 06/09/98 MLH
;  10/21/2002 Too few departments in class dept count rprf. MLH
;**************************************************************************************************
;*
;*  PROCESSES
;*
;**************************************************************************************************   
{
	Process to query out Whl_Sku, Whl_Class and Whl_Dept from the completed
	catalog data base and from this data create a file containing a record
	for each class,dept combination along with the number of skus within
	that class,dept.

}
proc NSKUperCLASSdept(itemf String, skudcf String, cldeftpl String, cdf String)
	var
		class, dept, sku, classlast, deptlast String
		srctb1 Table
		n longInt
		pname, nrec String
		tclde, tcsku TCursor
		qq Query
	endVar
	pname = "NSKUperCLASSdept:"

	; A. Build load tape item class dept database.
	message(selected, sname, pname, " Querying catalog sku dept class into ", skudcf, ".")

     qq = Query

	  ~itemf  | Whl_Sku | Whl_Class  | Whl_Dept |
				 | Check   | Check      | Check    |

	 EndQuery
 
   executeQBE(qq)
   srctb1.attach(":PRIV:ANSWER.DB")
	srctb1.copy(skudcf)


	; Sort sku class dept file by class and dept
	message(selected, sname, pname, " Sorting ", skudcf, " by class and dept.")
	srctb1.attach(skudcf)
	sort srctb1
		on "Whl_Class", "Whl_Dept"
	endSort

	; B. Create output class and error files
	srctb1.attach(cldeftpl)
   srctb1.empty()
	srctb1.copy(cdf)


	; Insert wholesaler skus with upc matches in match file with 1R match type
	srctb1.attach(skudcf)
	nrec = strVal(srctb1.nRecords())
	srctb1.unAttach()

	tclde.open(cdf)
	tcsku.open(skudcf)
	lock(tclde, "Exclusive", tcsku, "Exclusive")
	tclde.edit()
	tclde.home()
	tcsku.home()
	classcdproc = 0
	classC3_deptproc = 0
	itemcdproc = 0
	classlast = "x"
	deptlast = "x"
	scan tcsku:
		tcsku.fieldValue("Whl_Class", class)
		tcsku.fieldValue("Whl_Dept", dept)
		tcsku.fieldValue("Whl_Sku", sku)
		message(selected, sname, pname, "DataRec# ", tcsku.recNo(), " of ", nrec, ". Sku ", sku, " class ", class)

		itemcdproc = itemcdproc + 1
		switch
			case class <> classlast :
				n = 1
				classcdproc = classcdproc + 1
				classC3_deptproc = classC3_deptproc + 1
				tclde.insertAfterRecord()
				tclde.setFieldValue("Whl_Class", class)
				tclde.setFieldValue("Whl_Dept", dept)
				tclde.setFieldValue("Nsku", n)
				classlast = class
				deptlast = dept
			case class = classlast and dept <> deptlast :
				n = 1
				classC3_deptproc = classC3_deptproc + 1
				tclde.insertAfterRecord()
				tclde.setFieldValue("Whl_Class", class)
				tclde.setFieldValue("Whl_Dept", dept)
				tclde.setFieldValue("Nsku", n)
				deptlast = dept
			otherwise :
				n = n + 1
				tclde.setFieldValue("Nsku", n)
		endSwitch
		sleep(0)
	endScan
	tcsku.endedit()
	unlock(tclde, "Exclusive", tcsku, "Exclusive")
	tclde.close()
	tcsku.close()

	cat_rec_proc = itemcdproc
	inpt_rec_proc = classcdproc
	cat_rec_changed = classC3_deptproc


endProc

{
	Process to create a report by class of data in file containing a record
	for each class,dept along with the number of skus within that class,dept.
	The most referenced department in a class will be in field D1. This dept
	will then be used as the dept for that class in the Eagle Catalog class 
	data base.

}
proc CLASSdeptRPT(cdreptfmtf String, cdf String, cdreptf String)
	var
		class, dept, oldclass, deptfield, deptnskufield String
		n smallInt
      nsku longInt
		srctb1 Table
		pname, nrec String
		tcrpt, tccdf TCursor
	endVar
	pname = "CLASSdeptRPT:"

	; Sort class dept nsku file by class and nsku
	message(selected, sname, pname, " Sorting ", cdf, " by class and nsku.")
	srctb1.attach(cdf)
	sort srctb1
		on "Whl_Class", "Nsku" D
	endSort

	; Create output report file
	srctb1.attach(cdreptfmtf)
	srctb1.empty()
	srctb1.copy(cdreptf)

	; Insert dept nsku data by class into report
	srctb1.attach(cdf)
	nrec = strVal(srctb1.nRecords())
	srctb1.unAttach()

	tcrpt.open(cdreptf)
	tccdf.open(cdf)
	lock(tcrpt, "Exclusive", tccdf, "Exclusive")
	tcrpt.edit()
	tcrpt.home()
	tccdf.home()
	classcdproc = 1
	classC3_deptproc = 0
	oldclass = tccdf."Whl_Class"
	n = 1
	tcrpt.insertAfterRecord()
	scan tccdf:
		tccdf.fieldValue("Whl_Class", class)
		tccdf.fieldValue("Whl_Dept", dept)
		tccdf.fieldValue("Nsku", nsku)
		classC3_deptproc = classC3_deptproc + 1
		message(selected, sname, pname, "DataRec# ", tccdf.recNo(), " of ", nrec, ". Class ", class, ".")
		if oldclass < class then
			tcrpt.insertAfterRecord()
			oldclass = class
			classcdproc = classcdproc + 1
			n = 1
		endIf
		if n < 21 then    ; was 11 changed rpt file format to support 20 departments
			tcrpt.setFieldValue("Whl_Class", class) 
			deptfield = "D" + strVal(n)
			deptnskufield = deptfield + "nsku"
			tcrpt.setFieldValue(deptfield, dept) 
			tcrpt.setFieldValue(deptnskufield, nsku) 
			n = n + 1
		else
			msgInfo(class, "Too many departments!")
		endIf

		sleep(0)
	endScan
	tccdf.endedit()
	unlock(tcrpt, "Exclusive", tccdf, "Exclusive")
	tcrpt.close()
	tccdf.close()

	cat_rec_proc = classC3_deptproc
	inpt_rec_proc = classcdproc
	cat_rec_changed = classcdproc


endProc

proc makeecfi(template String, eccatf String, ltfif String, ecfinelines String, 
				 ecfiload String, errf String)
	var
 
	  storenumber String
	  fineline, fineline_cmp, offset String
	  i smallInt
	  tccode, tcout, tcin, tcerr Tcursor
	  srcTb1 Table
	  nrec, pname String
	  qq Query
	  tfiload String
	  tcfiload TCursor
	endVar
	pname = "makeecfi:"
	tfiload = ":PRIV:tfiload.db"

	; A. Build catalog fineline database.
	; Normalizing of fineline code coming in from item_raw will already have been
	; done in C1_itemproc.
	message(selected, sname, " Querying catalog fineline into ", ecfinelines, ".")

	  qq = Query

	  ~eccatf | Whl_Fineline  | Whl_Class |
				 | Check         | Check     |   

	 EndQuery
 
	executeQBE(qq)
	srctb1.attach(":PRIV:ANSWER.DB")
	srctb1.copy(ecfinelines)

	; A2, A3 and A4 Added 12/17/2009
	; A2. Build released description fineline database.
	; Normalizes fineline codes coming in from ltfi.
	message(selected, sname, " Querying catalog fineline into ", ecfinelines, ".")

	  qq = Query

	  ~ltfif | FI_FINELINE_CODE  | FI_FINELINE_NAME |
				| Check             | Check            |   

	 EndQuery
 
	executeQBE(qq)
	srctb1.attach(":PRIV:ANSWER.DB")
	srctb1.copy(tfiload)

	; A3. Correct fineline code if required.	
	tcfiload.open(tfiload)
	tcfiload.home()
	tcfiload.edit()
	scan tcfiload :
		code = tcfiload."FI_FINELINE_CODE"
		code = NormFinecode(code)
		tcfiload."FI_FINELINE_CODE" = code	
	endScan
	tcfiload.endEdit()
	tcfiload.close()

	; A4. Sort tables
	srctb1.attach(ecfinelines)
	sort srctb1
		On "Whl_Fineline"
	endSort
	srctb1.attach(tfiload)
	sort srctb1
		On "FI_FINELINE_CODE"
	endSort
	srctb1.unattach()

	; B. Create output fineline and error files
	srctb1.attach(template)
	srctb1.empty()
	srctb1.copy(ecfiload)

	inpt_rec_proc = 0
	cat_rec_changed = 0
	cat_rec_spec = 0            
	scan tccode:
		tccode.fieldValue("Whl_Fineline", fineline)

;		fineline_cmp = NormFinecode(fineline_cmp) ;commented out 12/17/2009

			while fineline_cmp < fineline and tcin.eot() = False

			endWhile
		endIf
		if fineline = fineline_cmp then
			cat_rec_changed = cat_rec_changed + 1
			tcout.insertAfterRecord()

			setdefault(tcout, "Fineline", fineline)

			intoout(tcin, "FI_FINELINE_NAME", tcout, 50, "Description")
			 
		else
         cat_rec_changed = cat_rec_changed + 1
         tcout.insertAfterRecord()

			; Alt_Ind X(01)

			; Fineline X(06)

			cat_rec_spec = cat_rec_spec + 1                           
      endIf
		sleep(0)
   endScan
	tcerr.endedit()
	tcout.endedit()
	unlock(tccode, "Exclusive", tcout, "Exclusive", tcin, "Exclusive", tcerr, "Exclusive")
   tcerr.close()
   tccode.close()
   tcout.close()
   tcin.close()

	message(selected, sname, pname, " Sorting ", ecfiload, " Fineline." )

proc xpertout(var ecat_err String, id String, txtf String, desc String,
					finDir String, remotefinDir String) Logical
	var
		xprtfmt, dbf, pathdbf, pathtxtf String
		batline	String 
	endvar
	
	; A. Define files
	xprtfmt = systemDir + id + "_EXP.DB"
	dbf = id + ".DB"
	pathdbf = finDir + dbf
	pathtxtf = finDir + txtf

	; B. Check for required files
	if chkfile(ecat_err, xprtfmt) = False then
		return False
	endIf
	if chkfile(ecat_err, pathdbf) = False then
		return False
	endIf

	; C. Export file
	time_start = Time()
	message(selected, sname, "Exporting ", dbf, " to ", txtf, ".")
	exportASCIIFix(pathdbf,pathtxtf,xprtfmt)
	time_complete = Time()
	process_desc = sname + selected + " Export " + desc + " data to txt file."
	BldHstReport(process_desc, pathdbf, pathdbf)

	; D. Copy to remote
	copyfile("Y", dbf, findir, remotefinDir)

	; E. Add txtf to zipf
	batline = systemdir + "PKZIP.EXE -a " + LocalFinal + FinalZIP + " " + pathtxtf
	callbat("CE", batline)

	return True
endProc

;|BeginMethod|#Library1|C3_deptproc|
;  C3_deptproc  Process LTDE data to ecat department file. Created 06/09/98 MLH
;**************************************************************************************************
;*
;*  PROCESSES
;*
;**************************************************************************************************   
; Department processing
proc makeecde(template String, eccatf String, ltdef String, ltdepts String, 
				 ltdeload String, errf String)
	var
 
	  dept, dept_cmp, offset String
     i smallInt
     tccode, tcout, tcin, tcerr Tcursor
     srcTb1 Table
     nrec, pname String
     qq Query

	endVar
	pname = "makeecde:"

	; A. Build catalog dept database.
   message(selected, sname, " Querying load tape department into ", ltdepts, ".")

     qq = Query

	  ~eccatf | Whl_Dept  |
			    | Check     |

    EndQuery
 
   executeQBE(qq)
   srctb1.attach(":PRIV:ANSWER.DB")
   srctb1.copy(ltdepts)

	; B. Create output department and error files
   srctb1.attach(template)
   srctb1.empty()
   srctb1.copy(ltdeload)

	; C. Insert department data into department file
   srctb1.attach(ltdepts)
   nrec = strVal(srctb1.nRecords())
   srctb1.unAttach()

   tccode.open(ltdepts)
   tcout.open(ltdeload)
   tcin.open(ltdef)
   tcerr.open(errf)
   lock(tccode, "Exclusive", tcout, "Exclusive", tcin, "Exclusive", tcerr, "Exclusive")
   tcout.edit()
   tcerr.edit()
   tcout.end()
   tcerr.end()
   tccode.home()
   tcin.home()
   cat_rec_proc = 0
   inpt_rec_proc = 0
   cat_rec_changed = 0
   cat_rec_spec = 0
   scan tccode:
		tccode.fieldValue("Whl_Dept", dept)
		tcin.fieldValue("DE_DEPARTMENT", dept_cmp)
		cat_rec_proc = cat_rec_proc + 1
      if dept_cmp < dept then
         while dept_cmp < dept and tcin.eot() = False
            tcin.nextRecord()
				tcin.fieldValue("DE_DEPARTMENT", dept_cmp)
            inpt_rec_proc = inpt_rec_proc + 1
         endWhile
      endIf
      if dept = dept_cmp then
         cat_rec_changed = cat_rec_changed + 1
         tcout.insertAfterRecord()
         message(selected, sname, pname," DERec# ", tccode.recNo(), " of ", nrec, ". Dept ", dept, ".")
         ;****************************************************     
         ; Fill supplied fields 
         ;****************************************************
			; Alt_Ind X(01) 
			setdefault(tcout, "Alt_Ind", "R")

			; Dept X(02)
			setdefault(tcout, "Dept", dept)

			; Desription X(50) 
			; Ace loadtape department file DE_DEPARTMENT_NAME X(32) 
			intoout(tcin, "DE_DEPARTMENT_NAME", tcout, 50, "Description")

			
		else

         cat_rec_changed = cat_rec_changed + 1
         tcout.insertAfterRecord()

			; Alt_Ind X(01) 
			setdefault(tcout, "Alt_Ind", "R")

			; Dept X(02) 
			setdefault(tcout, "Dept", dept)

			; Desription X(50) 
			setdefault(tcout, "Description", "UNDEFINED")			

         ; Report Not In File
         tcerr.insertAfterRecord()
         tcerr.setFieldValue("TypeCode", "Dept")
         tcerr.setfieldValue("RefCode", dept)
			tcerr.setfieldValue("Problem", "No name for Department")
			cat_rec_spec = cat_rec_spec + 1
							  
      endIf
      sleep(0)
   endScan
   tcerr.endedit()
   tcout.endedit()
   unlock(tccode, "Exclusive", tcout, "Exclusive", tcin, "Exclusive", tcerr, "Exclusive")
   tcerr.close()
   tccode.close()
   tcout.close()
   tcin.close()

	; F. Sort department output load data file again
   message(selected, sname, pname, " Sorting ", ltdeload, " by Store and Department." )
   srctb1.attach(ltdeload)
   sort srctb1
	  on "Dept"
   endSort

endproc

proc StdVndOut(template String, eccatf String, codef String, inptf String,

   var
	  code, code_cmp, newcode, vname, dateadded String
	  filedate Date
	  addr1, addr2, city, state, zip5, zip4, terms String

     ar Array[] String
     n  longInt
     en, posfob, termsize smallInt

     keepzeros, removezeros, numericpadzero, numericpadspace Logical
     tckv, tcerr, tccode, tcout, tcin Tcursor
     srcTb1 Table
     nrec, pname String
     qq Query
   endVar
	pname = "StdVndOut:"

	; A. Gather configuration data
	tcwhl.fieldValue("Cat_Raw_Date", filedate)
	dateadded = ReformatDate(filedate)

	; B. Build vendor database.
   message(selected, sname, " Querying load vendor codes into ", codef, ".")

    qq = Query

	 ~eccatf | Whl_Mfg_Code   |
				| Check          | 

    EndQuery

    executeQBE(qq)
    srctb1.attach(":PRIV:ANSWER.DB")
    srctb1.copy(codef)

	; C. Open vendor load File 
   srctb1.attach(template)
   srctb1.empty()
   srctb1.copy(outf)

	; D. Insert vendor data into vendor load file
   srctb1.attach(codef)
   nrec = strVal(srctb1.nRecords())
	srctb1.unAttach()

	tccode.open(codef)
   tcout.open(outf)
   tcin.open(inptf)
   lock(tccode, "Exclusive", tcout, "Exclusive", tcin, "Exclusive")
   tcout.edit()
   tcout.end()
   tccode.home()
   tcin.home()
   cat_rec_proc = 0
   inpt_rec_proc = 0
	cat_rec_changed = 0
	cat_rec_spec = 0
   keepzeros = True
   removezeros = False
   numericpadzero = True
   numericpadspace = False
   scan tccode:
		tccode.fieldValue("Whl_Mfg_Code", code)
		tcin.fieldValue("VmVendorCode", code_cmp)
      cat_rec_proc = cat_rec_proc + 1
      if code_cmp < code then
         while code_cmp < code and tcin.eot() = False
            tcin.nextRecord()
				tcin.fieldValue("VmVendorCode", code_cmp)
            inpt_rec_proc = inpt_rec_proc + 1
         endWhile
      endIf
      if code = code_cmp then
         tcout.insertAfterRecord()
         message(selected, sname, pname," VMRec# ", tccode.recNo(), " of ", nrec, ". Code ", code, ".")
         ;****************************************************     
         ; Fill supplied fields 
         ;****************************************************

			; Whl_Mfg_Code X(05)
			; LTVM VmVendorCode X(05)
			newcode = code.lTrim()    
			setdefault(tcout, "Whl_Mfg_Code", newcode) 

			; Date_Added 9(08) YYYYMMDD
			tcout.setFieldValue("Date_Added", dateadded)
			
			; Name X(35)

			vname = tcin."VmVendorName"

			setdefault(tcout, "Name", vname)
				
			; Address1 X(35)
			; LTVM VmAddr1 X(32)
			addr1 = tcin."VmAddr1"
			addr1 = addr1.lTrim()
			setdefault(tcout, "Address1", addr1)

			; Address2 X(35)
			; LTVM VmAddr2 X(32)
			addr2 = tcin."VmAddr2"
			addr2 = addr2.lTrim()
			setdefault(tcout, "Address2", addr2)
 
			; City X(25) 
			; LTVM VmCity X(16)
			city = tcin."VmCity"
			city = city.lTrim()
			setdefault(tcout, "City", city)

			; State X(2)
			; LTVM VmState X(2)
			state = tcin."VmState"
			state = state.lTrim()
			setdefault(tcout, "State", state)

			; Zip X(10)
			; LTVM VmZip5 X(5) and LTVM VmZip(4)
			zip5 = tcin."VmZip5"
			zip5 = zip5.lTrim()
			zip4 = tcin."VmZip4"
			zip4 = zip4.lTrim()
			if zip4 <> "" then
				setdefault(tcout, "Zip", zip5 + "-" + zip4)
			else
				setdefault(tcout, "Zip", zip5)
			endIf

			; Terms X(50)
			; LTVM VmTermsDesc X(15)
			terms = tcin."VmTermsDesc"
			terms = terms.lTrim()
			setdefault(tcout, "Terms", terms)

			cat_rec_changed = cat_rec_changed + 1
		else
			tcout.insertAfterRecord()

 			; Whl_Mfg_Code X(05)
			; LTVM VmVendorCode X(05)
			newcode = code.lTrim()    
			setdefault(tcout, "Whl_Mfg_Code", newcode) 

			; Date_Added 9(08) YYYYMMDD
			tcout.setFieldValue("Date_Added", dateadded)
			
			; Name X(35) 
			setdefault(tcout, "Name", "UNDEFINED")

			cat_rec_spec = cat_rec_spec + 1
			cat_rec_changed = cat_rec_changed + 1
		endIf
		sleep(0)
	endScan
	tcout.endedit()
	unlock(tccode, "Exclusive", tcout, "Exclusive", tcin, "Exclusive")
	tccode.close()
